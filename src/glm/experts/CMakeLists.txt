# GLM-specific expert targets; no dependency from Qwen into this directory.
if(STRATA_NATIVE_EXPERTS)
  include(cpu.cmake)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/vnni.cmake)
    include(vnni.cmake)
  endif()
  if(TARGET strata_glm_experts_vnni)
    include(bandwidth.cmake)
    include(r8.cmake)
  endif()
endif()
if(STRATA_ENABLE_CUDA)
  include(gpu.cmake)
  include(pin.cmake)
endif()

add_library(strata_glm_experts INTERFACE)
foreach(_target strata_glm_experts_cpu strata_glm_experts_gpu strata_glm_experts_vnni strata_glm_experts_r8)
  if(TARGET ${_target})
    target_link_libraries(strata_glm_experts INTERFACE ${_target})
  endif()
endforeach()
